  604    /**
  605     * Ensures that the output corresponding to {@code jsonish[start:end]} is a
  606     * valid JSON string that has the same meaning when parsed by Javascript
  607     * {@code eval}.
  608     * <ul>
  609     *   <li>Making sure that it is fully quoted with double-quotes.
  610     *   <li>Escaping any Javascript newlines : CR, LF, U+2028, U+2029
  611     *   <li>Escaping HTML special characters to allow it to be safely embedded
  612     *       in HTML {@code <script>} elements and XML {@code <!CDATA[...]]>}
  613     *       sections.
  614     *   <li>Rewrite hex, octal, and other escapes that are valid in Javascript
  615     *       but not in JSON.
  616     * </ul>
  617     * @param start inclusive
  618     * @param end   exclusive
  619     */

  621      boolean closed = false;
  622      for (int i = start; i < end; ++i) {
  623        char ch = jsonish.charAt(i);
  624        switch (ch) {
  625          case '\t': replace(i, i + 1, "\\t"); break;
  626          // Fixup newlines.
  627          case '\n': replace(i, i + 1, "\\n"); break;
  628          case '\r': replace(i, i + 1, "\\r"); break;
  629          // Not newlines in JSON but unparseable by JS eval.
  630          case '\u2028': replace(i, i + 1, "\\u2028"); break;
  631          case '\u2029': replace(i, i + 1, "\\u2029"); break;
  632          // String delimiting quotes that need to be converted : 'foo' -> "foo"
  633          // or internal quotes that might need to be escaped : f"o -> f\"o.
  634          case '"': case '\'':
  635            if (i == start) {
  636              if (ch == '\'') { replace(i, i + 1, '"'); }
  637            } else {
  638              if (i + 1 == end) {
  639                char startDelim = jsonish.charAt(start);
  640                if (startDelim != '\'') {
  641                  // If we're sanitizing a string whose start was inferred, then
  642                  // treat '"' as closing regardless.
  643                  startDelim = '"';
  644                }
  645                closed = startDelim == ch;
  646              }
  647              if (closed) {
  648                if (ch == '\'') { replace(i, i + 1, '"'); }
  649              } else if (ch == '"') {
  650                insert(i, '\\');
  651              }

  653            break;

  655          // literals so that the output can be embedded in HTML script elements
  656          // and in XML CDATA sections without affecting the parser state.
  657          // References:
  658          // https://www.w3.org/TR/html53/semantics-scripting.html#restrictions-for-contents-of-script-elements
  659          // https://www.w3.org/TR/html53/syntax.html#script-data-escaped-state

  664            // data escaped" state.

  667            // state.
  668            // Disallow </script, which ends a script block.

  670              break;
