  195     * Same as {@link JsonSanitizer#sanitize(Reader, Appendable)}, but allows to
  196     * set a custom maximum nesting depth.
  197     *
  198     * @param in JSON-like content.
  199     * @param out receives embeddable JSON.
  200     * @param maximumNestingDepth maximum nesting depth.
  201     * @throws IOException if reading {@code in} or writing {@code out} fails.
  202     */
  203    public static void sanitize(
  204        Reader in, Appendable out, int maximumNestingDepth) throws IOException {
  205      StringBuilder jsonish = new StringBuilder();
  206      char[] buf = new char[8192];
  207      for (int n; (n = in.read(buf)) >= 0;) {
  208        jsonish.append(buf, 0, n);
  209      }
  210      JsonSanitizer s = new JsonSanitizer(
  211          jsonish.toString(), maximumNestingDepth);
  212      s.sanitize();
  213      out.append(s.toCharSequence());
  214    }
