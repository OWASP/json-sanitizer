   24  echo and make sure that the current POM release number is max.

   26  # Pick a release version
   27  export VERSION="$(mvn -q \
   28       -Dexec.executable="echo" \
   29       -Dexec.args='${project.version}' \

   31       org.codehaus.mojo:exec-maven-plugin:1.3.1:exec \

   35       | perl -pe 's/^(\d+\.\d+)(?:\.(\d+))?.*/$1.".".($2||0)/e')";

   37  export NEW_DEV_VERSION="$(echo "$VERSION" \
   38       | perl -pe 's/^(\d+\.\d+)(?:\.(\d+))?.*/$1.".".($2+1)."-SNAPSHOT"/e')";

   40  echo "

   59  
   60  # Update the version
