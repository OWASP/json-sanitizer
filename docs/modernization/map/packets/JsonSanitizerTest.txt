   35    private static void assertSanitized(String golden, String input, int maximumNestingDepth) {
   36      String actual = sanitize(input, maximumNestingDepth);
   37      assertEquals(input, golden, actual);
   38      if (actual.equals(input)) {
   39        assertSame(input, input, actual);
   40      }
   41    }
