   75    @Test
   76    public static final void testNameGenerator() {
   77      NameGenerator ng = new NameGenerator();
   78      assertEquals("a", ng.next());
   79      assertEquals("b", ng.next());
   80      assertEquals("c", ng.next());
   81      for (int i = 30; --i >= 0;) { ng.next(); }
   82      assertEquals("H", ng.next());
   83      Set<String> shortNames = new HashSet<String>();
   84      for (int i = 0; i < 1000; ++i) { shortNames.add(ng.next()); }
   85      assertTrue(shortNames.contains("ar"));
   86      assertFalse(shortNames.contains("as"));
   87      assertTrue(shortNames.contains("au"));
   88      assertTrue(shortNames.contains("dn"));
   89      assertFalse(shortNames.contains("do"));
   90      assertTrue(shortNames.contains("dp"));
   91      assertTrue(shortNames.contains("ie"));
   92      assertFalse(shortNames.contains("if"));
   93      assertTrue(shortNames.contains("ig"));
   94      assertTrue(shortNames.contains("ih"));
   95      assertTrue(shortNames.contains("ii"));
   96      assertTrue(shortNames.contains("ij"));
   97      assertTrue(shortNames.contains("ik"));
   98      assertTrue(shortNames.contains("il"));
   99      assertTrue(shortNames.contains("im"));
  100      assertFalse(shortNames.contains("in"));
  101      assertTrue(shortNames.contains("io"));

  104    @Test
  105    public static final void testReservedKeywordsSorted() {
  106      String[][] keywords = EvalMinifier.RESERVED_KEYWORDS;
  107      for (int len = 0; len < keywords.length; ++len) {
  108        String[] row = keywords[len];
  109        for (int i = 0; i < row.length; ++i) {
  110          assertEquals(row[i], len, row[i].length());
  111          if (i > 0) {
  112            assertTrue(row[i - 1] + " < " + row[i], row[i - 1].compareTo(row[i]) < 0);
  113          }
  114          assertTrue(row[i], Arrays.binarySearch(row, row[i]) >= 0);
  115        }
  116      }
  117    }
